#### 🎯 学习方法

**核心概念理解 (30分钟)**

```
假设检验的量化语境:

H0 (零假设): 策略期望收益 = 0 (策略无效)
H1 (备择假设): 策略期望收益 > 0 (策略有效)

t统计量 = (样本均值 - 假设值) / (标准误)
       = (策略平均收益 - 0) / (收益标准差/√n)

p值: 如果H0为真，观察到当前结果或更极端结果的概率

决策规则:
- p < 0.05: 拒绝H0，策略可能有效
- p >= 0.05: 无法拒绝H0，策略可能无效
```

**手工计算练习 (15分钟)**

问题: 某策略回测20天，日收益均值0.5%，标准差2%。检验策略是否显著有效。

解答步骤:
```
t = (0.5% - 0) / (2%/√20) = 0.005 / 0.00447 = 1.12

查t分布表，df=19:
- 单侧p值 ≈ 0.14

结论: p > 0.05，无法拒绝H0，策略有效性不显著
需要更长的回测数据或更低的波动
```

**关键洞察 (15分钟)**

```
为什么你的策略回测15%年化可能没用？

场景A: 夏普1.5，回测2年
- t = 1.5 × √2 = 2.12
- p ≈ 0.017 < 0.05 ✓ 显著

场景B: 夏普1.5，回测3个月
- t = 1.5 × √0.25 = 0.75
- p ≈ 0.23 > 0.05 ✗ 不显著

结论: 短期高收益 ≠ 有效策略，样本量是关键
```

```python
"""
策略显著性检验模块
用于验证策略收益是否统计显著
"""
import numpy as np
import pandas as pd
from scipy import stats
from typing import Tuple, Dict, Optional


class StrategySignificanceTester:
    """
    策略统计显著性检验器

    核心功能:
    1. t检验：策略收益是否显著不为0
    2. 夏普比率显著性检验
    3. 置信区间估计
    """

    def __init__(self, risk_free_rate: float = 0.03):
        """
        初始化

        Args:
            risk_free_rate: 年化无风险利率，默认3%
        """
        self.risk_free_rate = risk_free_rate

    def t_test_returns(
        self,
        returns: np.ndarray,
        null_hypothesis: float = 0.0
    ) -> Dict[str, float]:
        """
        对收益序列进行t检验

        Args:
            returns: 日收益率序列
            null_hypothesis: 零假设下的期望收益

        Returns:
            {
                't_statistic': t统计量,
                'p_value': p值,
                'mean_return': 样本均值,
                'std_return': 样本标准差,
                'is_significant': 是否显著(p<0.05)
            }
        """
        n = len(returns)
        mean_ret = np.mean(returns)
        std_ret = np.std(returns, ddof=1)  # 样本标准差

        # 计算t统计量
        se = std_ret / np.sqrt(n)  # 标准误
        t_stat = (mean_ret - null_hypothesis) / se

        # 计算p值（单侧检验）
        p_value = 1 - stats.t.cdf(t_stat, df=n-1)

        return {
            't_statistic': t_stat,
            'p_value': p_value,
            'mean_return': mean_ret,
            'std_return': std_ret,
            'n_samples': n,
            'is_significant': p_value < 0.05,
            'standard_error': se
        }

    def sharpe_ratio_test(
        self,
        returns: np.ndarray,
        periods_per_year: int = 252
    ) -> Dict[str, float]:
        """
        检验夏普比率是否统计显著

        原理: 夏普比率 × √T 服从t分布

        Args:
            returns: 日收益率序列
            periods_per_year: 年化周期数，日数据=252

        Returns:
            {
                'sharpe_ratio': 夏普比率,
                'annualized_sharpe': 年化夏普,
                't_statistic': t统计量,
                'p_value': p值,
                'is_significant': 是否显著
            }
        """
        n = len(returns)

        # 计算夏普比率
        excess_returns = returns - self.risk_free_rate / periods_per_year
        sharpe = np.mean(excess_returns) / np.std(excess_returns, ddof=1)

        # 年化夏普
        annual_sharpe = sharpe * np.sqrt(periods_per_year)

        # 夏普比率的t检验
        # t = Sharpe × √T
        t_stat = sharpe * np.sqrt(n)
        p_value = 1 - stats.t.cdf(t_stat, df=n-1)

        return {
            'sharpe_ratio': sharpe,
            'annualized_sharpe': annual_sharpe,
            't_statistic': t_stat,
            'p_value': p_value,
            'is_significant': p_value < 0.05,
            'n_samples': n
        }

    def confidence_interval(
        self,
        returns: np.ndarray,
        confidence: float = 0.95
    ) -> Tuple[float, float]:
        """
        计算策略收益的置信区间

        Args:
            returns: 收益率序列
            confidence: 置信水平

        Returns:
            (下限, 上限)
        """
        n = len(returns)
        mean_ret = np.mean(returns)
        se = stats.sem(returns)  # 标准误

        # t分布临界值
        t_critical = stats.t.ppf((1 + confidence) / 2, df=n-1)

        margin = t_critical * se
        return (mean_ret - margin, mean_ret + margin)


# 便捷函数
def quick_strategy_report(returns: np.ndarray, strategy_name: str = "Strategy") -> str:
    """
    快速生成策略显著性报告

    Args:
        returns: 收益率序列
        strategy_name: 策略名称

    Returns:
        格式化的报告字符串
    """
    tester = StrategySignificanceTester()

    t_result = tester.t_test_returns(returns)
    sharpe_result = tester.sharpe_ratio_test(returns)
    ci_lower, ci_upper = tester.confidence_interval(returns)

    report = f"""
{'='*50}
{strategy_name} 统计显著性报告
{'='*50}

样本信息:
  样本数: {t_result['n_samples']}
  平均日收益: {t_result['mean_return']:.4%}
  日收益标准差: {t_result['std_return']:.4%}

t检验 (H0: 收益 = 0):
  t统计量: {t_result['t_statistic']:.4f}
  p值: {t_result['p_value']:.4f}
  是否显著 (p<0.05): {'✓ 是' if t_result['is_significant'] else '✗ 否'}

夏普比率检验:
  夏普比率: {sharpe_result['sharpe_ratio']:.4f}
  年化夏普: {sharpe_result['annualized_sharpe']:.4f}
  t统计量: {sharpe_result['t_statistic']:.4f}
  p值: {sharpe_result['p_value']:.4f}
  是否显著: {'✓ 是' if sharpe_result['is_significant'] else '✗ 否'}

95% 置信区间:
  日收益: [{ci_lower:.4%}, {ci_upper:.4%}]
  年化收益: [{ci_lower*252:.2%}, {ci_upper*252:.2%}]

{'='*50}
结论: {'策略统计显著，可考虑实盘' if t_result['is_significant'] and sharpe_result['is_significant'] else '策略不显著，需更多数据或优化'}
{'='*50}
"""
    return report
```

```python
import unittest
import numpy as np
from risk_management.statistical_tests.strategy_significance import (
    StrategySignificanceTester, quick_strategy_report
)


class TestStrategySignificance(unittest.TestCase):

    def setUp(self):
        """测试前准备"""
        np.random.seed(42)
        self.tester = StrategySignificanceTester()

    def test_t_test_with_zero_returns(self):
        """测试零收益情况"""
        returns = np.random.normal(0, 0.02, 100)  # 均值=0
        result = self.tester.t_test_returns(returns)

        self.assertGreater(result['p_value'], 0.05)  # 应该不显著
        self.assertFalse(result['is_significant'])

    def test_t_test_with_positive_returns(self):
        """测试正收益情况"""
        returns = np.random.normal(0.001, 0.02, 252)  # 正收益
        result = self.tester.t_test_returns(returns)

        # 大概率显著（取决于随机种子）
        self.assertIsInstance(result['t_statistic'], float)
        self.assertIsInstance(result['p_value'], float)

    def test_sharpe_ratio_calculation(self):
        """测试夏普比率计算"""
        returns = np.random.normal(0.001, 0.02, 252)
        result = self.tester.sharpe_ratio_test(returns)

        self.assertIsInstance(result['sharpe_ratio'], float)
        self.assertIsInstance(result['annualized_sharpe'], float)

    def test_confidence_interval(self):
        """测试置信区间"""
        returns = np.random.normal(0.001, 0.02, 100)
        lower, upper = self.tester.confidence_interval(returns)

        self.assertLess(lower, upper)
        self.assertLess(lower, np.mean(returns))
        self.assertGreater(upper, np.mean(returns))


if __name__ == '__main__':
    unittest.main()
```

### 周三（3/19）| 多重比较校正：测试100个因子后的真相

#### 📖 学习资源

**首选资源**
- [StatQuest: Multiple Comparisons](https://www.youtube.com/watch?v=YK9A0FA-VXw) (8分钟)
- [Multiple Testing - 知乎](https://zhuanlan.zhihu.com/p/133536485)

**深入理解**
- 《All of Statistics》Wasserman, Chapter 12
- [Benjamini-Hochberg方法](https://www.jstor.org/stable/2346101)

#### 🎯 学习方法

**核心概念理解 (30分钟)**

```
多重比较问题:

如果你测试100个随机策略，每个都用p<0.05作为显著标准：
- 预期会有 100 × 0.05 = 5个策略"显著"
- 这5个完全是假阳性！

这就是p-hacking，也是大多数量化新手亏钱的原因。

解决方案:

1. Bonferroni校正（最保守）
   α_adjusted = α / m
   其中m是测试次数

   例子: 测试100个因子，α=0.05
   α_adjusted = 0.05/100 = 0.0005

   缺点: 过于保守，容易漏掉真信号

2. Benjamini-Hochberg (FDR控制)
   控制"假发现率"而非"假阳性率"
   更适合因子挖掘场景

   步骤:
   a) 对所有p值排序: p(1) ≤ p(2) ≤ ... ≤ p(m)
   b) 找到最大的k使得 p(k) ≤ (k/m) × α
   c) 拒绝H0(1)到H0(k)
```

**手工计算练习 (15分钟)**

问题: 测试5个因子，p值分别为 [0.01, 0.03, 0.04, 0.1, 0.2]，α=0.05

Bonferroni校正:
```
α_adjusted = 0.05/5 = 0.01
只有p=0.01的因子显著
```

Benjamini-Hochberg:
```
排序: 0.01, 0.03, 0.04, 0.1, 0.2
i=1: 0.01 ≤ (1/5)×0.05 = 0.01 ✓
i=2: 0.03 ≤ (2/5)×0.05 = 0.02 ✗

最大k=1，只有第一个因子显著
```

```python
from statsmodels.stats.multitest import multipletests


class MultipleComparisonCorrection:
    """
    多重比较校正

    用于因子挖掘、策略筛选等场景
    """

    @staticmethod
    def bonferroni_correction(
        p_values: np.ndarray,
        alpha: float = 0.05
    ) -> Dict:
        """
        Bonferroni校正

        Args:
            p_values: p值数组
            alpha: 显著性水平

        Returns:
            {
                'corrected_alpha': 校正后的α,
                'is_significant': 各因子是否显著的布尔数组,
                'n_significant': 显著因子数
            }
        """
        m = len(p_values)
        corrected_alpha = alpha / m
        is_significant = p_values < corrected_alpha

        return {
            'corrected_alpha': corrected_alpha,
            'is_significant': is_significant,
            'n_significant': np.sum(is_significant),
            'method': 'Bonferroni'
        }

    @staticmethod
    def benjamini_hochberg(
        p_values: np.ndarray,
        alpha: float = 0.05
    ) -> Dict:
        """
        Benjamini-Hochberg FDR控制

        Args:
            p_values: p值数组
            alpha: 显著性水平

        Returns:
            {
                'adjusted_pvalues': 调整后的p值,
                'is_significant': 各因子是否显著的布尔数组,
                'n_significant': 显著因子数
            }
        """
        reject, adjusted_pvals, _, _ = multipletests(
            p_values, alpha=alpha, method='fdr_bh'
        )

        return {
            'adjusted_pvalues': adjusted_pvals,
            'is_significant': reject,
            'n_significant': np.sum(reject),
            'method': 'Benjamini-Hochberg'
        }

    @staticmethod
    def factor_screening_report(
        factor_names: list,
        p_values: np.ndarray,
        sharpe_ratios: np.ndarray,
        alpha: float = 0.05
    ) -> str:
        """
        生成因子筛选报告

        Args:
            factor_names: 因子名称列表
            p_values: p值数组
            sharpe_ratios: 夏普比率数组
            alpha: 显著性水平

        Returns:
            格式化的报告
        """
        # 应用两种校正方法
        bonf = MultipleComparisonCorrection.bonferroni_correction(p_values, alpha)
        bh = MultipleComparisonCorrection.benjamini_hochberg(p_values, alpha)

        report = f"""
{'='*70}
多重比较校正报告
{'='*70}

测试概况:
  总因子数: {len(factor_names)}
  显著性水平 α: {alpha}

原始结果 (未校正):
  显著因子数: {np.sum(p_values < alpha)}
  假阳性预期: {len(factor_names) * alpha:.1f}个

Bonferroni校正 (控制族错误率):
  校正后 α: {bonf['corrected_alpha']:.6f}
  显著因子数: {bonf['n_significant']}
  保守程度: 高（可能漏掉真信号）

Benjamini-Hochberg (控制假发现率):
  显著因子数: {bh['n_significant']}
  方法: 控制FDR在{alpha:.0%}以下
  保守程度: 中等（推荐用于因子挖掘）

{'='*70}
因子详细结果
{'='*70}
"""

        # 按p值排序
        sorted_idx = np.argsort(p_values)

        for i in sorted_idx[:10]:  # 只显示前10个
            report += f"\n{factor_names[i]:<20} "
            report += f"p={p_values[i]:.4f} "
            report += f"Sharpe={sharpe_ratios[i]:.2f} "
            report += f"BH:{'✓' if bh['is_significant'][i] else '✗'} "
            report += f"Bonf:{'✓' if bonf['is_significant'][i] else '✗'}"

        report += f"\n{'='*70}\n"
        report += f"结论: 建议使用BH方法，发现{bh['n_significant']}个显著因子\n"
        report += f"{'='*70}\n"

        return report
```

### 周四（3/20）| 因子回归分析：剥离市场风险后的真实alpha

#### 📖 学习资源

**首选 - 视频理解**
- [StatQuest: Linear Regression](https://www.youtube.com/watch?v=aq8VU5KLmkY) (8分钟)

**深入 - 代码实践**

- [Fama-French Data](https://mba.tuck.dartmouth.edu/pages/faculty/ken.french/data_library.html)

#### 🎯 学习方法

**核心概念 (30分钟)**

```
线性回归在量化中的核心作用:

策略收益 = α + β₁×市场因子 + β₂×市值因子 + β₃×价值因子 + ε

其中:
- α (alpha): 剥离风险因子后的超额收益，这是真正的"能力"
- β (beta): 对各风险因子的暴露
- ε (残差): 随机噪音

关键洞察:
- 高 raw return 可能是承担了高风险
- 高 alpha 才是真正有价值的
- 如果 α 不显著，你的策略只是在暴露已知风险因子
```

**Fama-French三因子模型 (15分钟)**

```
R_strategy - R_f = α + β_mkt × (R_mkt - R_f)
                   + β_smb × SMB
                   + β_hml × HML
                   + ε

其中:
- R_mkt - R_f: 市场超额收益
- SMB (Small Minus Big): 市值因子，小盘减大盘
- HML (High Minus Low): 价值因子，高BP减低BP

A股实践:
- 市场因子: 沪深300收益 - 无风险利率
- SMB: 小市值组合 - 大市值组合
- HML: 高BP组合 - 低BP组合
```

#### 💻 实践任务

**任务**: 实现因子回归分析模块

**Step 1 - 创建回归分析模块 (40分钟)**

```python
"""
因子回归分析模块
用于剥离风险因子，提取真实alpha
"""
import numpy as np
import pandas as pd
import statsmodels.api as sm
from typing import Dict, Optional, Tuple


class FactorRegressionAnalyzer:
    """
    因子回归分析器

    使用Fama-French模型剥离风险因子
    """

    def __init__(self, risk_free_rate: float = 0.03):
        self.risk_free_rate = risk_free_rate

    def fama_french_regression(
        self,
        strategy_returns: np.ndarray,
        market_returns: np.ndarray,
        smb: Optional[np.ndarray] = None,
        hml: Optional[np.ndarray] = None,
        periods_per_year: int = 252
    ) -> Dict:
        """
        Fama-French三因子回归

        Args:
            strategy_returns: 策略日收益
            market_returns: 市场日收益
            smb: 市值因子收益（可选）
            hml: 价值因子收益（可选）
            periods_per_year: 年化周期

        Returns:
            回归结果字典
        """
        # 计算超额收益
        strategy_excess = strategy_returns - self.risk_free_rate / periods_per_year
        market_excess = market_returns - self.risk_free_rate / periods_per_year

        # 构建自变量矩阵
        X = pd.DataFrame({'MKT': market_excess})

        if smb is not None:
            X['SMB'] = smb
        if hml is not None:
            X['HML'] = hml

        X = sm.add_constant(X)  # 添加截距项（alpha）

        # 运行回归
        model = sm.OLS(strategy_excess, X)
        results = model.fit(cov_type='HAC', cov_kwds={'maxlags': 5})  # Newey-West标准误

        # 提取结果
        output = {
            'alpha': results.params['const'],
            'alpha_tstat': results.tvalues['const'],
            'alpha_pvalue': results.pvalues['const'],
            'alpha_annualized': results.params['const'] * periods_per_year,
            'alpha_significant': results.pvalues['const'] < 0.05,
            'beta_mkt': results.params['MKT'],
            'beta_mkt_tstat': results.tvalues['MKT'],
            'r_squared': results.rsquared,
            'adj_r_squared': results.rsquared_adj,
            'resid_std': np.std(results.resid),
            'sharpe_of_alpha': (results.params['const'] * np.sqrt(periods_per_year)) /
                              np.std(results.resid),
            'n_obs': len(strategy_returns),
            'summary': str(results.summary())
        }

        if smb is not None:
            output['beta_smb'] = results.params['SMB']
            output['beta_smb_tstat'] = results.tvalues['SMB']

        if hml is not None:
            output['beta_hml'] = results.params['HML']
            output['beta_hml_tstat'] = results.tvalues['HML']

        return output

    def generate_regression_report(self, results: Dict) -> str:
        """
        生成回归分析报告

        Args:
            results: 回归结果字典

        Returns:
            格式化的报告
        """
        report = f"""
{'='*70}
Fama-French回归分析报告
{'='*70}

样本信息:
  观测数: {results['n_obs']}
  R²: {results['r_squared']:.4f}
  调整R²: {results['adj_r_squared']:.4f}

Alpha (超额收益):
  日Alpha: {results['alpha']:.6f}
  年化Alpha: {results['alpha_annualized']:.2%}
  t统计量: {results['alpha_tstat']:.4f}
  p值: {results['alpha_pvalue']:.4f}
  是否显著: {'✓ 是' if results['alpha_significant'] else '✗ 否'}
  Alpha夏普: {results['sharpe_of_alpha']:.4f}

风险因子暴露:
  β_市场: {results['beta_mkt']:.4f} (t={results['beta_mkt_tstat']:.2f})
"""

        if 'beta_smb' in results:
            report += f"  β_市值: {results['beta_smb']:.4f} (t={results['beta_smb_tstat']:.2f})\n"
        if 'beta_hml' in results:
            report += f"  β_价值: {results['beta_hml']:.4f} (t={results['beta_hml_tstat']:.2f})\n"

        report += f"""
残差标准差: {results['resid_std']:.4f}

{'='*70}
结论: {'策略有显著Alpha' if results['alpha_significant'] else '策略Alpha不显著，收益来自风险暴露'}
{'='*70}
"""
        return report


def create_synthetic_ff_factors(
    market_returns: np.ndarray,
    seed: int = 42
) -> Tuple[np.ndarray, np.ndarray]:
    """
    创建合成的Fama-French因子（用于演示）

    实际使用时应该使用真实的FF因子数据
    """
    np.random.seed(seed)

    # SMB: 小盘通常在市场上涨时表现更好，但波动更大
    smb = np.random.normal(0.0002, 0.008, len(market_returns)) + market_returns * 0.1

    # HML: 价值因子与市场有一定负相关
    hml = np.random.normal(0.0001, 0.007, len(market_returns)) - market_returns * 0.05

    return smb, hml
```

#### 📖 学习资源

**首选 - 视频理解**

#### 🎯 学习方法

**核心概念 (30分钟)**

```
Bootstrap的核心思想:

"数据本身就是对总体的最好估计"

步骤:
1. 从原始数据中有放回地抽样，生成B个bootstrap样本
2. 对每个样本计算统计量（如夏普比率）
3. 用这B个统计量的分布作为真实分布的估计

优势:
- 不依赖正态分布假设
- 适用于小样本
- 可以估计任何统计量的置信区间

在量化中的应用:
- 估计策略夏普比率的置信区间
- 计算最大回撤的分布
- 评估策略参数的稳健性
```

#### 💻 实践任务

**任务**: 实现Bootstrap分析模块

**Step 1 - 添加Bootstrap方法 (30分钟)**

在`strategy_significance.py`中添加：

```python
from sklearn.utils import resample


class BootstrapAnalyzer:
    """
    Bootstrap分析器

    用于不依赖分布假设的稳健统计推断
    """

    @staticmethod
    def bootstrap_sharpe(
        returns: np.ndarray,
        n_bootstrap: int = 10000,
        confidence: float = 0.95,
        periods_per_year: int = 252
    ) -> Dict:
        """
        Bootstrap估计夏普比率的置信区间

        Args:
            returns: 收益率序列
            n_bootstrap: bootstrap次数
            confidence: 置信水平
            periods_per_year: 年化周期

        Returns:
            {
                'sharpe_point_estimate': 点估计,
                'sharpe_bootstrap_mean': bootstrap均值,
                'sharpe_std': bootstrap标准差,
                'confidence_interval': 置信区间,
                'bootstrap_sharpes': 所有bootstrap夏普值
            }
        """
        n = len(returns)

        # 计算点估计
        point_sharpe = np.mean(returns) / np.std(returns, ddof=1) * np.sqrt(periods_per_year)

        # Bootstrap
        bootstrap_sharpes = []
        for _ in range(n_bootstrap):
            # 有放回抽样
            sample = resample(returns, n_samples=n)
            sharpe = np.mean(sample) / np.std(sample, ddof=1) * np.sqrt(periods_per_year)
            bootstrap_sharpes.append(sharpe)

        bootstrap_sharpes = np.array(bootstrap_sharpes)

        # 计算置信区间（百分位法）
        alpha = 1 - confidence
        ci_lower = np.percentile(bootstrap_sharpes, alpha/2 * 100)
        ci_upper = np.percentile(bootstrap_sharpes, (1 - alpha/2) * 100)

        return {
            'sharpe_point_estimate': point_sharpe,
            'sharpe_bootstrap_mean': np.mean(bootstrap_sharpes),
            'sharpe_std': np.std(bootstrap_sharpes),
            'confidence_interval': (ci_lower, ci_upper),
            'is_significant': ci_lower > 0,  # 置信区间完全在0以上
            'bootstrap_sharpes': bootstrap_sharpes
        }

    @staticmethod
    def bootstrap_max_drawdown(
        returns: np.ndarray,
        n_bootstrap: int = 5000
    ) -> Dict:
        """
        Bootstrap估计最大回撤的分布

        Args:
            returns: 收益率序列
            n_bootstrap: bootstrap次数

        Returns:
            {
                'mdd_point_estimate': 点估计,
                'mdd_mean': bootstrap均值,
                'mdd_median': bootstrap中位数,
                'mdd_95th': 95%分位数（最坏情况）
            }
        """
        n = len(returns)

        # 计算点估计
        cum_returns = np.cumprod(1 + returns) - 1
        running_max = np.maximum.accumulate(cum_returns)
        drawdowns = cum_returns - running_max
        point_mdd = np.min(drawdowns)

        # Bootstrap
        bootstrap_mdds = []
        for _ in range(n_bootstrap):
            sample = resample(returns, n_samples=n)
            cum = np.cumprod(1 + sample) - 1
            running_max = np.maximum.accumulate(cum)
            dd = cum - running_max
            bootstrap_mdds.append(np.min(dd))

        bootstrap_mdds = np.array(bootstrap_mdds)

        return {
            'mdd_point_estimate': point_mdd,
            'mdd_mean': np.mean(bootstrap_mdds),
            'mdd_median': np.median(bootstrap_mdds),
            'mdd_95th': np.percentile(bootstrap_mdds, 5),  # 最坏5%情况
            'bootstrap_mdds': bootstrap_mdds
        }


def comprehensive_strategy_analysis(returns: np.ndarray, strategy_name: str = "Strategy") -> str:
    """
    综合策略分析：t检验 + Bootstrap

    Args:
        returns: 收益率序列
        strategy_name: 策略名称

    Returns:
        综合分析报告
    """
    # t检验
    tester = StrategySignificanceTester()
    t_result = tester.t_test_returns(returns)
    sharpe_result = tester.sharpe_ratio_test(returns)

    # Bootstrap
    bootstrap = BootstrapAnalyzer()
    bs_sharpe = bootstrap.bootstrap_sharpe(returns)
    bs_mdd = bootstrap.bootstrap_max_drawdown(returns)

    report = f"""
{'='*70}
{strategy_name} 综合分析报告
{'='*70}

一、t检验结果
  t统计量: {t_result['t_statistic']:.4f}
  p值: {t_result['p_value']:.4f}
  是否显著: {'✓' if t_result['is_significant'] else '✗'}

二、夏普比率分析
  点估计: {sharpe_result['annualized_sharpe']:.4f}
  Bootstrap均值: {bs_sharpe['sharpe_bootstrap_mean']:.4f}
  Bootstrap标准差: {bs_sharpe['sharpe_std']:.4f}
  95%置信区间: [{bs_sharpe['confidence_interval'][0]:.4f}, {bs_sharpe['confidence_interval'][1]:.4f}]
  Bootstrap显著: {'✓' if bs_sharpe['is_significant'] else '✗'}

三、最大回撤分析
  点估计: {bs_mdd['mdd_point_estimate']:.2%}
  Bootstrap均值: {bs_mdd['mdd_mean']:.2%}
  最坏5%情况: {bs_mdd['mdd_95th']:.2%}

{'='*70}
综合结论: {'策略稳健，建议实盘' if t_result['is_significant'] and bs_sharpe['is_significant'] else '策略不稳健，需优化或更多数据'}
{'='*70}
"""
    return report
```

```python
"""
策略质量门禁系统
================

只有满足所有统计标准的策略才能进入实盘候选池
"""
import numpy as np
from typing import Dict, List, Optional
from dataclasses import dataclass

from risk_management.statistical_tests.strategy_significance import (
    StrategySignificanceTester, MultipleComparisonCorrection,
    BootstrapAnalyzer, FactorRegressionAnalyzer
)


@dataclass
class StrategyQualityGate:
    """策略质量标准"""
    min_sharpe: float = 1.0
    min_samples: int = 252
    max_pvalue: float = 0.05
    min_alpha_annualized: float = 0.05  # 5%年化alpha
    max_mdd_95th: float = -0.30  # 最坏5%情况回撤不超过30%
    require_alpha_significant: bool = True


class StrategyGateKeeper:
    """
    策略门禁 keeper

    对策略进行全面的统计检验，只有通过所有检验的策略才能进入实盘池
    """

    def __init__(self, gate: Optional[StrategyQualityGate] = None):
        self.gate = gate or StrategyQualityGate()
        self.tester = StrategySignificanceTester()
        self.bootstrap = BootstrapAnalyzer()
        self.regression = FactorRegressionAnalyzer()

    def evaluate_strategy(
        self,
        strategy_name: str,
        returns: np.ndarray,
        market_returns: Optional[np.ndarray] = None
    ) -> Dict:
        """
        全面评估策略质量

        Args:
            strategy_name: 策略名称
            returns: 策略收益序列
            market_returns: 市场收益序列（用于FF回归）

        Returns:
            评估结果字典
        """
        results = {
            'strategy_name': strategy_name,
            'passed': True,
            'checks': {},
            'details': {}
        }

        # 检查1: 样本量
        n_samples = len(returns)
        results['checks']['sample_size'] = n_samples >= self.gate.min_samples
        results['details']['sample_size'] = {
            'actual': n_samples,
            'required': self.gate.min_samples
        }

        # 检查2: t检验显著性
        t_result = self.tester.t_test_returns(returns)
        results['checks']['t_test_significant'] = t_result['is_significant']
        results['details']['t_test'] = t_result

        # 检查3: 夏普比率
        sharpe_result = self.tester.sharpe_ratio_test(returns)
        results['checks']['sharpe_sufficient'] = (
            sharpe_result['annualized_sharpe'] >= self.gate.min_sharpe
        )
        results['details']['sharpe'] = sharpe_result

        # 检查4: Bootstrap夏普置信区间
        bs_sharpe = self.bootstrap.bootstrap_sharpe(returns)
        results['checks']['bootstrap_significant'] = bs_sharpe['is_significant']
        results['details']['bootstrap_sharpe'] = bs_sharpe

        # 检查5: 最大回撤
        bs_mdd = self.bootstrap.bootstrap_max_drawdown(returns)
        results['checks']['mdd_acceptable'] = (
            bs_mdd['mdd_95th'] >= self.gate.max_mdd_95th
        )
        results['details']['mdd'] = bs_mdd

        # 检查6: Alpha显著性（如果有市场数据）
        if market_returns is not None:
            ff_result = self.regression.fama_french_regression(
                returns, market_returns
            )
            results['checks']['alpha_significant'] = (
                ff_result['alpha_significant'] and
                ff_result['alpha_annualized'] >= self.gate.min_alpha_annualized
            )
            results['details']['alpha'] = ff_result
        else:
            results['checks']['alpha_significant'] = True  # 跳过

        # 综合判断
        results['passed'] = all(results['checks'].values())

        return results

    def generate_gate_report(self, results: Dict) -> str:
        """生成门禁报告"""
        report = f"""
{'='*70}
策略质量门禁报告: {results['strategy_name']}
{'='*70}

综合结果: {'✅ 通过 - 可进入实盘池' if results['passed'] else '❌ 未通过 - 需优化或更多数据'}

各项检查:
"""
        for check_name, passed in results['checks'].items():
            status = '✅' if passed else '❌'
            report += f"  {status} {check_name}\n"

        # 详细数据
        report += f"\n{'='*70}\n详细数据\n{'='*70}\n"

        if 'sharpe' in results['details']:
            s = results['details']['sharpe']
            report += f"\n夏普比率:\n"
            report += f"  年化夏普: {s['annualized_sharpe']:.4f} (要求>={self.gate.min_sharpe})\n"
            report += f"  p值: {s['p_value']:.4f}\n"

        if 'bootstrap_sharpe' in results['details']:
            bs = results['details']['bootstrap_sharpe']
            report += f"\nBootstrap夏普:\n"
            report += f"  95% CI: [{bs['confidence_interval'][0]:.4f}, {bs['confidence_interval'][1]:.4f}]\n"

        if 'mdd' in results['details']:
            m = results['details']['mdd']
            report += f"\n最大回撤:\n"
            report += f"  最坏5%: {m['mdd_95th']:.2%} (要求>={self.gate.max_mdd_95th:.0%})\n"

        if 'alpha' in results['details']:
            a = results['details']['alpha']
            report += f"\nAlpha分析:\n"
            report += f"  年化Alpha: {a['alpha_annualized']:.2%}\n"
            report += f"  是否显著: {'是' if a['alpha_significant'] else '否'}\n"

        report += f"\n{'='*70}\n"

        return report

    def batch_evaluate(
        self,
        strategies: Dict[str, np.ndarray],
        market_returns: Optional[np.ndarray] = None
    ) -> List[Dict]:
        """
        批量评估多个策略

        Args:
            strategies: {策略名: 收益序列} 字典
            market_returns: 市场收益序列

        Returns:
            各策略评估结果列表
        """
        results = []
        for name, returns in strategies.items():
            result = self.evaluate_strategy(name, returns, market_returns)
            results.append(result)

        return results


# 便捷函数
def screen_strategies_for_production(
    strategies: Dict[str, np.ndarray],
    market_returns: Optional[np.ndarray] = None,
    gate: Optional[StrategyQualityGate] = None
) -> List[str]:
    """
    筛选可实盘的策略

    Args:
        strategies: 策略字典
        market_returns: 市场收益
        gate: 质量标准（使用默认或自定义）

    Returns:
        通过筛选的策略名称列表
    """
    keeper = StrategyGateKeeper(gate)
    results = keeper.batch_evaluate(strategies, market_returns)

    passed_strategies = [
        r['strategy_name'] for r in results if r['passed']
    ]

    print(f"策略筛选结果:")
    print(f"  总策略数: {len(strategies)}")
    print(f"  通过筛选: {len(passed_strategies)}")
    print(f"  通过率: {len(passed_strategies)/len(strategies):.1%}")
    print(f"\n可实盘策略: {passed_strategies}")

    return passed_strategies
```

1. **p值的真正含义**
   ```
   "p值不是'假设为真的概率'，而是'如果假设为真，
   观察到当前结果或更极端结果的概率'。

   量化语境: p=0.03意味着如果策略真的无效，
   只有3%的概率会看到这么好的回测结果。"
   ```

2. **为什么需要多重比较校正**
   ```
   "测试100个随机策略，预期会有5个p<0.05的'假阳性'。
   如果不校正，你会把这5个当成有效策略，实盘就亏钱。"
   ```

3. **Bootstrap的优势**
   ```
   "传统t检验假设收益服从正态分布，但A股收益有肥尾。
   Bootstrap直接从数据抽样，不依赖分布假设，更稳健。"
   ```
